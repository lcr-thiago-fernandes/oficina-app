        +string Motivo
        +Guid? OrdemServicoId
        +DateTimeOffset CriadoEm
    }

    class Sku {
        <<vo>>
        +string Valor
    }

    class TipoMovimentacao {
        <<enum>>
        Entrada
        Saida
    }

    Peca "1" *-- "0..*" MovimentacaoEstoque : registra
    Peca *-- Sku : identificada por
    MovimentacaoEstoque ..> TipoMovimentacao

    note for Peca "Invariantes:<br/>1. Saldo NUNCA fica negativo<br/>2. Quantidade > 0 em todas movimentações<br/>3. SaldoAtual atualizado na MESMA transação<br/>4. Saída pode referenciar uma OS<br/>5. Movimentações imutáveis após gravadas<br/>6. SKU único"

**Política central do contexto:** `RegistrarSaida` falha com `SaldoInsuficienteException` se `quantidade > SaldoAtual`.

## 4. Agregado `OrdemDeServico` (Núcleo) ⭐

```mermaid
classDiagram
    class OrdemDeServico {
        <<root>>
        +Guid Id
        +long Numero
        +Guid ClienteId
        +Guid VeiculoId
        +StatusOrdemDeServico Status
        +decimal TotalServicos
        +decimal TotalPecas
        +decimal TotalGeral
        +DateTimeOffset? OrcamentoAprovadoEm
        +DateTimeOffset? IniciadaEm
        +DateTimeOffset? FinalizadaEm
        +TimeSpan? DuracaoExecucao
        +Criar(clienteId, veiculoId, obs) OrdemDeServico
        +IniciarDiagnostico()
        +AdicionarItemServico(servicoId, nome, preco, qtd)
        +AdicionarItemPeca(pecaId, nome, preco, qtd)
        +RemoverItemServico(itemId)
        +RemoverItemPeca(itemId)
        +EnviarOrcamentoParaAprovacao()
        +Aprovar()
        +Rejeitar()
        +IniciarExecucao()
        +Finalizar()
        +Entregar()
    }

    class ItemServico {
        +Guid Id
        +Guid ServicoId
        +string ServicoNome
        +decimal PrecoSnapshot
        +int Quantidade
        +decimal Subtotal
    }

    class ItemPeca {
        +Guid Id
        +Guid PecaId
        +string PecaNome
        +decimal PrecoSnapshot
        +int Quantidade
        +decimal Subtotal
    }

    class StatusOrdemDeServico {
        <<enum>>
        Recebida
        EmDiagnostico
        AguardandoAprovacao
        EmExecucao
        Finalizada
        Entregue
        Cancelada
    }

    OrdemDeServico "1" *-- "0..*" ItemServico : contém
    OrdemDeServico "1" *-- "0..*" ItemPeca : contém
    OrdemDeServico ..> StatusOrdemDeServico

    note for OrdemDeServico "Invariantes:<br/>1. Máquina de estados — transições só por métodos da raiz<br/>2. Itens editáveis SOMENTE antes de EmExecucao<br/>3. Snapshot de nome e preço preservado nos itens<br/>4. Aprovação só em AguardandoAprovacao<br/>5. Execução exige OrcamentoAprovadoEm preenchido<br/>6. Numero é BIGSERIAL gerado pelo banco<br/>7. Domain events disparados em transições"
```

## 5. Agregado `Usuario` (Auth — módulo técnico)

Não é um bounded context de negócio, mas estruturalmente segue o mesmo padrão de agregado.

```mermaid
classDiagram
    class Usuario {
        <<root>>
        +Guid Id
        +Username Username
        +Senha Senha
        +Perfil Perfil
        +bool Ativo
        +bool PrecisaTrocarSenha
        +Criar(username, senha, perfil) Usuario
        +CriarParaBootstrap(username, senha) Usuario
        +Autenticar(textoPuro) bool
        +TrocarSenha(novaSenha)
        +Inativar()
        +Ativar()
    }

    class Username {
        <<vo>>
        +string Valor
    }

    class Senha {
        <<vo>>
        +string Hash
        +DeTextoPuro(texto) Senha
        +DeHashExistente(hash) Senha
        +Verificar(textoPuro) bool
    }

    class Perfil {
        <<enum>>
        Admin
        Atendente
    }

    Usuario *-- Username
    Usuario *-- Senha
    Usuario ..> Perfil

    note for Usuario "Invariantes:<br/>1. Username 3-50 chars [a-z0-9._]<br/>2. Senha mínima 8 chars; BCrypt cost 12<br/>3. Autenticar lança se Inativo<br/>4. Bootstrap marca PrecisaTrocarSenha=true"
```

## Resumo: regras de design dos agregados

| Regra | Aplicação no projeto |
|---|---|
| **Raiz é única porta de entrada** | Construtores das entidades internas (`Veiculo.Criar`, `ItemServico.Criar`) são `internal` |
| **Pequeno por padrão** | Cada agregado tem apenas o que precisa coexistir transacionalmente |
| **Referências por ID entre agregados** | `OrdemDeServico` referencia `ClienteId`/`VeiculoId`/`PecaId`/`ServicoId` como `Guid`, não navegação |
| **Snapshot quando necessário** | `ItemServico` e `ItemPeca` guardam cópia de `nome` e `preco` no momento da inclusão |
| **Eventos de domínio para integração** | OS dispara `ExecucaoIniciadaEvent`; handler cross-context baixa estoque |
| **Invariantes na raiz** | Saldo de Peça, máquina de estados de OS, unicidade de placa por cliente |
